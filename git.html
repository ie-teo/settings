git diff - изменения которые не были закоммичены

git log -p - показывает изменения всех файлов

git log -p -n1 - сколько коммитов вывести, тут один

git checkout fc6a - возвращаемся на предыдущий коммит, 4 цифры коммита

git log --oneline - показывает хеш и комментарий к коммиту в виде аккуратного списка

git branch -v - показывает данные о текущем коммите

git branch feature - создание новой ветки

git checkout -b fix - создать ветку и сразу переключиться на нее

git reflog --date=iso - показывает коммиты с датой

git log --pretty=format:'%h %cd | %s%d | [%an]' - красиво выводит список коммитов

git branch - показывает ветки и гит

git branch new-feature - создается ветка new-feature

git checkout - переключение веток

git checkout -b another - создание новой ветки и сразу переключение на нее

git branch -d another - удаление ветки (нужно быть на другой ветке прежде чем удалять)

git branch -m main - переименует текущую ветку

git log --pretty=format:'%h %cd | %s%d | [%an]' - настройка алиасов

git log --pretty=my - выведется удобный список

git config --global format.pretty my - команда задает формат вывода по умолчанию

git log - теперь выводится список по алиасу

git log --grep 111 - показывает все коммиты где упоминается слово 111

git log --grep 111 feature - будет искать коммиты в ветке feature

git log -Gtitle - выведет коммиты со словом title

git log -L 3,6:1.txt - выводит все коммиты где есть файл 1.txt с 3 по 6 строку

git log -L '<title>','<\/title>':hello.html - покажет изменения от тега title до /title

git log -1 -p - выводит последний коммит и показывает что было изменено

git checkout -- file5.txt - откатить конкретный файл до последнего коммита

git log - выводит хеши коммитов

git checkout 32d5 - переключиться на определенный коммит по первым символам хеша

git checkout master - вернуться на последний коммит

git commit -am "добавление в индекс и сразу коммит одной командой" - добавление и коммит одной командой

git config --global alias.commitall '!git add .;git commit' - создание алиаса для add и commit

git branch -v - показывает все ветки

git branch feature - создаем новую ветку feature

git checkout feature - переключение на ветку feature

git checkout -b feature - создает и сразу переключает на новую ветку

git show --pretty=fuller - подробная информация о коммитах

git log --pretty=oneline - красивый вывод всех коммитов

git log --oneline --no-decorate - короткий вывод коммитов без декораций

git log --pretty=format:'%h %cd | %s%d' --date=short - еще один красивый вывод коммитов
